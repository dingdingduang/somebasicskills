            if (STBasicLibs_TestKeyEvent3.isDown()) {
                NetworkingMsgInitialization.sendToServer(new SendTestToServer());

//                applyTimedArmorToEntity(sp1, 300, temprandomdouble, OP_ADDITION, false, "guard2");
            }

            if (STBasicLibs_TestKeyEvent1.isDown()) {
                ServerPlayerEntity sp1 = ConvertClientPlayerToServerPlayer(cp1);
//                //printout status map
                HashMap<String, Double> tempStatusMap = getSPlayerValue2BaseMultiplierMap().get(sp1);
                for (String attrName: tempStatusMap.keySet()) {
                    double attrAmount = Math.abs(tempStatusMap.get(attrName));
                    if (attrAmount > 0.001 || Math.round(attrAmount) > 0) {
                        printInGameMsg(String.format("%s: %f ", attrName, tempStatusMap.get(attrName)));

package net.dingdingduang.somebasicskills.networking.packet.key;

import com.google.common.collect.Lists;
import com.google.common.collect.Maps;
import net.minecraft.network.FriendlyByteBuf;
import net.minecraft.server.MinecraftServer;
import net.minecraft.server.level.ServerPlayer;
import net.minecraftforge.network.NetworkEvent;

import java.util.ArrayList;
import java.util.HashMap;
import java.util.function.Supplier;

import static net.dingdingduang.somebasicskills.util.fileio.FileReadWriteMethods.*;
import static net.dingdingduang.somebasicskills.globalmethods.GeneralMethods.getMinecraftServerInstance;
import static net.dingdingduang.somebasicskills.globalvalues.GlobalServerPlayerValues.getSPlayerKeyCombo2SkillID;

public class SendKeyComboSettingToServer {
    private HashMap<ArrayList<Integer>, String> KeyCombo2SkillID;

    public SendKeyComboSettingToServer(HashMap<ArrayList<Integer>, String> keyCombo2SkillID) {
        this.KeyCombo2SkillID = keyCombo2SkillID;
    }

    public SendKeyComboSettingToServer(FriendlyByteBuf buf) {
        this.KeyCombo2SkillID = buf.readMap(Maps::newHashMapWithExpectedSize, (buf2) -> buf2.readCollection(Lists::newArrayListWithCapacity, FriendlyByteBuf::readInt), FriendlyByteBuf::readUtf);
    }

    public void toBytes(FriendlyByteBuf buf) {
        buf.writeMap(this.KeyCombo2SkillID, (buf2, arrIntList) -> buf2.writeCollection(arrIntList, FriendlyByteBuf::writeInt), FriendlyByteBuf::writeUtf);
    }

    public boolean handle(Supplier<NetworkEvent.Context> ctx) {
        ctx.get().enqueueWork(() -> {
            //do shiet on server side
            ServerPlayer sp1 = ctx.get().getSender();

            MinecraftServer mcServer = getMinecraftServerInstance();
            if (mcServer == null) { return; }

            getSPlayerKeyCombo2SkillID().put(sp1, this.KeyCombo2SkillID);
            KeycomboPlayernameFileWriteTo(mcServer, sp1);
        });

        return true;
    }
}
